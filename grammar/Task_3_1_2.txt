# CS325 — TASK 1.2.1
# FINAL LL(k) GRAMMAR WITH ARRAYS + FIRST & FOLLOW SETS (ε = epsilon)
# Grammar aligned with current mccomp.cpp implementation

======================================================
PART A — GRAMMAR (left recursion removed; LL(k); precedence enforced)
======================================================

program ::= extern_list decl_list
          | decl_list

extern_list ::= extern extern_list'
extern_list' ::= extern extern_list'
               | ε

extern ::= "extern" type_spec IDENT "(" params ")" ";"

decl_list ::= decl decl_list'
decl_list' ::= decl decl_list'
             | ε

decl ::= var_decl
       | fun_decl

# scalar and array global declarations
var_decl ::= var_type IDENT ";"
           | arr_type IDENT array_suffix ";"

type_spec ::= "void"
            | var_type

var_type ::= "int" | "float" | "bool"

# arrays only for int and float
arr_type ::= "int" | "float"

fun_decl ::= type_spec IDENT "(" params ")" block

params ::= param_list
         | "void"
         | ε

param_list ::= param param_list'
param_list' ::= "," param param_list'
              | ε

# scalar and array parameters
param ::= var_type IDENT
        | arr_type IDENT array_suffix

block ::= "{" local_decls stmt_list "}"

local_decls ::= local_decl local_decls'
              | ε
local_decls' ::= local_decl local_decls'
               | ε

# scalar and array local declarations
local_decl ::= var_type IDENT ";"
             | arr_type IDENT array_suffix ";"

# array declarations: 1D, 2D, 3D with fixed sizes
array_suffix ::= "[" INT_LIT "]"
               | "[" INT_LIT "]" "[" INT_LIT "]"
               | "[" INT_LIT "]" "[" INT_LIT "]" "[" INT_LIT "]"

stmt_list ::= stmt stmt_list'
            | ε
stmt_list' ::= stmt stmt_list'
             | ε

stmt ::= array_assign_stmt
       | expr_stmt
       | block
       | if_stmt
       | while_stmt
       | return_stmt

# array assignment statements (1D, 2D, 3D access)
array_assign_stmt ::= IDENT array_access_suffix "=" expr ";"

expr_stmt ::= expr ";"
            | ";"

while_stmt ::= "while" "(" expr ")" stmt

if_stmt ::= "if" "(" expr ")" block else_stmt

else_stmt ::= "else" block
            | ε

return_stmt ::= "return" ";"
              | "return" expr ";"

# ---- Expressions (lowest → highest precedence) ----
expr ::= IDENT "=" expr
       | rval

rval ::= or_expr

or_expr    ::= and_expr or_tail
or_tail    ::= "||" and_expr or_tail
             | ε

and_expr   ::= eq_expr and_tail
and_tail   ::= "&&" eq_expr and_tail
             | ε

eq_expr    ::= rel_expr eq_tail
eq_tail    ::= "==" rel_expr eq_tail
             | "!=" rel_expr eq_tail
             | ε

rel_expr   ::= add_expr rel_tail
rel_tail   ::= "<=" add_expr rel_tail
             | "<"  add_expr rel_tail
             | ">=" add_expr rel_tail
             | ">"  add_expr rel_tail
             | ε

add_expr   ::= mul_expr add_tail
add_tail   ::= "+" mul_expr add_tail
             | "-" mul_expr add_tail
             | ε

mul_expr   ::= unary mul_tail
mul_tail   ::= "*" unary mul_tail
             | "/" unary mul_tail
             | "%" unary mul_tail
             | ε

unary      ::= "-" unary
             | "!" unary
             | primary

primary    ::= "(" expr ")"
             | IDENT primary_suffix
             | INT_LIT
             | FLOAT_LIT
             | BOOL_LIT

# function calls and array accesses at same precedence
primary_suffix ::= "(" args ")"              # function call on IDENT
                 | array_access_suffix       # IDENT followed by indices
                 | ε

# array element access: 1D, 2D, or 3D
array_access_suffix ::= "[" expr "]"
                      | "[" expr "]" "[" expr "]"
                      | "[" expr "]" "[" expr "]" "[" expr "]"

args ::= arg_list
       | ε

arg_list ::= expr arg_tail
arg_tail ::= "," expr arg_tail
           | ε


======================================================
PART B — FIRST SETS (for ALL nonterminals)
======================================================

FIRST(program)        = { "extern", "void", "int", "float", "bool" }

FIRST(extern_list)    = { "extern" }
FIRST(extern_list')   = { "extern", ε }
FIRST(extern)         = { "extern" }

FIRST(decl_list)      = { "void", "int", "float", "bool" }
FIRST(decl_list')     = { "void", "int", "float", "bool", ε }
FIRST(decl)           = { "void", "int", "float", "bool" }

FIRST(var_decl)       = { "int", "float", "bool" }
FIRST(type_spec)      = { "void", "int", "float", "bool" }
FIRST(var_type)       = { "int", "float", "bool" }
FIRST(arr_type)       = { "int", "float" }

FIRST(fun_decl)       = { "void", "int", "float", "bool" }

FIRST(params)         = { "int", "float", "bool", "void", ε }
FIRST(param_list)     = { "int", "float", "bool" }
FIRST(param_list')    = { ",", ε }
FIRST(param)          = { "int", "float", "bool" }

FIRST(block)          = { "{" }
FIRST(local_decls)    = { "int", "float", "bool", ε }
FIRST(local_decls')   = { "int", "float", "bool", ε }
FIRST(local_decl)     = { "int", "float", "bool" }

FIRST(array_suffix)   = { "[" }

FIRST(stmt_list)      = { "{", "if", "while", "return", IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT, ";", ε }
FIRST(stmt_list')     = { "{", "if", "while", "return", IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT, ";", ε }
FIRST(stmt)           = { "{", "if", "while", "return", IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT, ";" }

FIRST(array_assign_stmt) = { IDENT }

FIRST(expr_stmt)      = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT, ";" }
FIRST(while_stmt)     = { "while" }
FIRST(if_stmt)        = { "if" }
FIRST(else_stmt)      = { "else", ε }
FIRST(return_stmt)    = { "return" }

FIRST(expr)           = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(rval)           = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }

FIRST(or_expr)        = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(or_tail)        = { "||", ε }

FIRST(and_expr)       = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(and_tail)       = { "&&", ε }

FIRST(eq_expr)        = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(eq_tail)        = { "==", "!=", ε }

FIRST(rel_expr)       = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(rel_tail)       = { "<=", "<", ">=", ">", ε }

FIRST(add_expr)       = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(add_tail)       = { "+", "-", ε }

FIRST(mul_expr)       = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(mul_tail)       = { "*", "/", "%", ε }

FIRST(unary)          = { "-", "!", "(", IDENT, INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(primary)        = { "(", IDENT, INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(primary_suffix) = { "(", "[", ε }
FIRST(array_access_suffix) = { "[" }

FIRST(args)           = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT, ε }
FIRST(arg_list)       = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(arg_tail)       = { ",", ε }


======================================================
PART C — FOLLOW SETS (headline & ε-nonterminals)
======================================================

FOLLOW(program)       = { $ }

# Lists / tails around top-level structure
FOLLOW(extern_list)   = { "void", "int", "float", "bool" }
FOLLOW(extern_list')  = { "void", "int", "float", "bool" }
FOLLOW(decl_list)     = { $ }
FOLLOW(decl_list')    = { $ }

# Params / args
FOLLOW(params)        = { ")" }
FOLLOW(param_list)    = { ")" }
FOLLOW(param_list')   = { ")" }

FOLLOW(args)          = { ")" }
FOLLOW(arg_list)      = { ")" }
FOLLOW(arg_tail)      = { ")" }

# Blocks & statements
FOLLOW(block)         = { "{", "if", "while", "return", IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT, ";", "}", $ }
FOLLOW(local_decls)   = { "{", "if", "while", "return", IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT, ";", "}" }
FOLLOW(local_decls')  = FOLLOW(local_decls)
FOLLOW(local_decl)    = FOLLOW(local_decls)
FOLLOW(stmt_list)     = { "}" }
FOLLOW(stmt_list')    = { "}" }
FOLLOW(stmt)          = FOLLOW(stmt_list)

# Arrays in declarations
FOLLOW(array_suffix)  = { ";", ")", "," }

# Expression contexts (used in conditions, returns, expr_stmt, args)
FOLLOW(expr)          = { ")", ";", ",", "}" }
FOLLOW(rval)          = FOLLOW(expr)

# Main expression nonterminals
FOLLOW(or_expr)       = FOLLOW(expr)
FOLLOW(or_tail)       = FOLLOW(or_expr)          # = { ")", ";", ",", "}" }

FOLLOW(and_expr)      = { "||" } ∪ FOLLOW(or_expr)
FOLLOW(and_tail)      = { "||" } ∪ FOLLOW(or_expr)

FOLLOW(eq_expr)       = { "&&" } ∪ FOLLOW(and_expr)
FOLLOW(eq_tail)       = { "&&", "||" } ∪ FOLLOW(or_expr)

FOLLOW(rel_expr)      = { "==", "!=" } ∪ FOLLOW(eq_expr)
FOLLOW(rel_tail)      = { "==", "!=", "&&", "||" } ∪ FOLLOW(or_expr)

FOLLOW(add_expr)      = { "<", "<=", ">", ">=" } ∪ FOLLOW(rel_expr)
FOLLOW(add_tail)      = { "<", "<=", ">", ">=", "==", "!=", "&&", "||" } ∪ FOLLOW(or_expr)

FOLLOW(mul_expr)      = { "+", "-" } ∪ FOLLOW(add_expr)
FOLLOW(mul_tail)      = { "+", "-", "<", "<=", ">", ">=", "==", "!=", "&&", "||" } ∪ FOLLOW(or_expr)

FOLLOW(unary)         = FOLLOW(mul_expr)
FOLLOW(primary)       = FOLLOW(unary)

# Call / array postfix after IDENT:
FOLLOW(primary_suffix)  = { "*", "/", "%", "+", "-", "<", "<=", ">", ">=", "==", "!=", "&&", "||", ")", ";", ",", "}" }

FOLLOW(var_decl)      = FOLLOW(decl_list)
FOLLOW(local_decl)    = FOLLOW(local_decls)
FOLLOW(param)         = { ")", "," }


======================================================
PART D — LL(k) DESIGN NOTES AND LINK TO CODE
======================================================

1. Arrays only for int and float

Scalar types stay as

    var_type ::= "int" | "float" | "bool"

Array element types use

    arr_type ::= "int" | "float"

and only arr_type appears with array_suffix. So declarations follow

    var_decl   ::= var_type IDENT ";"
                 | arr_type IDENT array_suffix ";"

    local_decl ::= var_type IDENT ";"
                 | arr_type IDENT array_suffix ";"

    param      ::= var_type IDENT
                 | arr_type IDENT array_suffix

This matches ParseDecl, ParseLocalDecl and ParseParam. BOOL_TOK never reaches ParseArrayDimsDecl, so bool arrays do not appear.

2. Fixed 1D, 2D, 3D arrays with explicit sizes

Declaration suffix

    array_suffix ::= "[" INT_LIT "]"
                   | "[" INT_LIT "]" "[" INT_LIT "]"
                   | "[" INT_LIT "]" "[" INT_LIT "]" "[" INT_LIT "]"

matches ParseArrayDimsDecl, which reads one to three "[" INT_LIT "]" groups and enforces a minimum of one dimension and a maximum of three. Declarations such as

    int a[5];
    int m[4][4];
    float t[3][3][4];

parse, while forms without sizes or with four dimensions fail in the parser.

3. Array access at the same precedence level as function calls

Expressions use

    primary ::= "(" expr ")"
              | IDENT primary_suffix
              | INT_LIT
              | FLOAT_LIT
              | BOOL_LIT

    primary_suffix ::= "(" args ")"
                     | array_access_suffix
                     | ε

    array_access_suffix ::= "[" expr "]"
                          | "[" expr "]" "[" expr "]"
                          | "[" expr "]" "[" expr "]" "[" expr "]"

So after IDENT the parser chooses between a call or an array access based on one token of lookahead

- "(" for a function call
- "[" for array element access
- a token from FOLLOW(primary_suffix) to end the postfix

This matches ParsePrimary, which on IDENT checks CurTok.type for LPAR or LBOX and builds either CallExprAST or ArrayAccessAST. ParseArrayIndices bounds the number of indices to three, which matches array_access_suffix.

4. Array assignment statements

Array assignments have their own statement form

    array_assign_stmt ::= IDENT array_access_suffix "=" expr ";"

and stmt has array_assign_stmt as first alternative. This matches ParseStmt logic

- on IDENT followed by "[" the parser scans all index groups
- looks ahead one token beyond the last "]"
- if this token is "=" it calls ParseArrayAssignStmt
- ParseArrayAssignStmt parses IDENT + indices as an ArrayAccessAST, then an "=" expr ";", and builds ArrayAssignAST

Expressions do not contain array assignments. So the grammar and code agree on placement of assignment for arrays.

5. Expression precedence and IDENT "=" expr vs rval

Expression parsing uses

    expr ::= IDENT "=" expr
           | rval

with the standard LL(2) decision. ParseExper matches this

- if the current token is IDENT it peeks one token ahead
- if that token is ASSIGN it parses IDENT "=" expr and returns AssignAST
- otherwise it pushes the peeked token back, restores CurTok to the IDENT, and parses rval

Every other expression level (or_expr, and_expr, eq_expr, rel_expr, add_expr, mul_expr, unary, primary) matches the usual precedence ladder and aligns with ParseOrExpr down to ParsePrimary.

6. FIRST/FOLLOW and LL(k) checks around primary_suffix

primary_suffix has

    FIRST(primary_suffix) = { "(", "[", ε }

and the FOLLOW set includes all binary operators and closing tokens

    FOLLOW(primary_suffix)  =
        { "*", "/", "%", "+", "-", "<", "<=", ">", ">=",
          "==", "!=", "&&", "||", ")", ";", ",", "}" }

FIRST(primary_suffix) and FOLLOW(primary_suffix) are disjoint, so the parser has an LL(1) decision at this position

- "(" → call
- "[" → array_access_suffix
- any symbol in FOLLOW(primary_suffix) → ε

This reflects ParsePrimary, which uses CurTok to decide between call, array access, or plain variable reference.

7. Link to code structure

This grammar matches key points in mccomp.cpp

- Lexer: tokens IDENT, INT_LIT, FLOAT_LIT, BOOL_LIT, all keyword tokens, all operators, and LBOX/RBOX match the terminal set
- Declarations and arrays
  - ParseDecl parses global scalars as GlobVarDeclAST
  - global arrays as ArrayDeclAST with dimensions from ParseArrayDimsDecl
  - functions as FunctionDeclAST with FunctionPrototypeAST holding ParamAST entries
- Local arrays: ParseLocalDecl builds VarDeclAST for scalars and ArrayDeclAST for int and float arrays
- Parameters: ParseParam attaches dimensions only for INT_TOK and FLOAT_TOK and passes them to ParamAST
- Array access in expressions: ParsePrimary plus ParseArrayIndices build ArrayAccessAST
- Array assignment: ParseArrayAssignStmt builds ArrayAssignAST matching array_assign_stmt
- Function calls: CallExprAST matches the call branch in primary_suffix

The LL(k) decisions in the document match the points in code that use extra lookahead (IDENT "=" expr versus rval, and statement-level array assignment versus vanilla expr_stmt). The grammar now describes exactly the programs that this compiler accepts for MiniC with 1D, 2D and 3D arrays of int and float.
