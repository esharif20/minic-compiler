# CS325 — TASK 1.2.1
# FINAL LL(k) GRAMMAR WITH ARRAYS + FIRST & FOLLOW SETS (ε = epsilon)
# Uses the SAME nonterminal names as Figure 1 (incl. `rval`), with layered sub-NONTERMINALS for precedence.
# Extended to support 1D, 2D, 3D arrays (int and float) in declarations and element access.

======================================================
PART A — GRAMMAR (left recursion removed; LL(k); precedence enforced)
======================================================

program ::= extern_list decl_list
          | decl_list

extern_list ::= extern extern_list'
extern_list' ::= extern extern_list'
               | ε

extern ::= "extern" type_spec IDENT "(" params ")" ";"

decl_list ::= decl decl_list'
decl_list' ::= decl decl_list'
             | ε

decl ::= var_decl
       | fun_decl

# scalar and array global declarations
var_decl ::= var_type IDENT ";"
           | arr_type IDENT array_suffix ";"

type_spec ::= "void"
            | var_type

var_type ::= "int" | "float" | "bool"

# arrays only for int and float
arr_type ::= "int" | "float"

fun_decl ::= type_spec IDENT "(" params ")" block

params ::= param_list
         | "void"
         | ε

param_list ::= param param_list'
param_list' ::= "," param param_list'
              | ε

# scalar and array parameters
param ::= var_type IDENT
        | arr_type IDENT array_suffix

block ::= "{" local_decls stmt_list "}"

local_decls ::= local_decl local_decls'
              | ε
local_decls' ::= local_decl local_decls'
               | ε

# scalar and array local declarations
local_decl ::= var_type IDENT ";"
             | arr_type IDENT array_suffix ";"

# array declarations: 1D, 2D, 3D with fixed sizes
array_suffix ::= "[" INT_LIT "]"
               | "[" INT_LIT "]" "[" INT_LIT "]"
               | "[" INT_LIT "]" "[" INT_LIT "]" "[" INT_LIT "]"

stmt_list ::= stmt stmt_list'
            | ε
stmt_list' ::= stmt stmt_list'
             | ε

stmt ::= expr_stmt
       | block
       | if_stmt
       | while_stmt
       | return_stmt

expr_stmt ::= expr ";"
            | ";"

while_stmt ::= "while" "(" expr ")" stmt

if_stmt ::= "if" "(" expr ")" block else_stmt

else_stmt ::= "else" block
            | ε

return_stmt ::= "return" ";"
              | "return" expr ";"

# ---- Expressions (lowest → highest precedence) ----
expr ::= IDENT "=" expr
       | rval

rval ::= or_expr

or_expr    ::= and_expr or_tail
or_tail    ::= "||" and_expr or_tail
             | ε

and_expr   ::= eq_expr and_tail
and_tail   ::= "&&" eq_expr and_tail
             | ε

eq_expr    ::= rel_expr eq_tail
eq_tail    ::= "==" rel_expr eq_tail
             | "!=" rel_expr eq_tail
             | ε

rel_expr   ::= add_expr rel_tail
rel_tail   ::= "<=" add_expr rel_tail
             | "<"  add_expr rel_tail
             | ">=" add_expr rel_tail
             | ">"  add_expr rel_tail
             | ε

add_expr   ::= mul_expr add_tail
add_tail   ::= "+" mul_expr add_tail
             | "-" mul_expr add_tail
             | ε

mul_expr   ::= unary mul_tail
mul_tail   ::= "*" unary mul_tail
             | "/" unary mul_tail
             | "%" unary mul_tail
             | ε

unary      ::= "-" unary
             | "!" unary
             | primary

primary    ::= "(" expr ")"
             | IDENT primary_tail
             | INT_LIT
             | FLOAT_LIT
             | BOOL_LIT

# function calls and array accesses at the same precedence
primary_tail ::= "(" args ")" primary_tail
               | "[" expr "]" primary_tail
               | ε

args ::= arg_list
       | ε

arg_list ::= expr arg_tail
arg_tail ::= "," expr arg_tail
           | ε

FIRST(program)        = { "extern", "void", "int", "float", "bool" }

FIRST(extern_list)    = { "extern" }
FIRST(extern_list')   = { "extern", ε }
FIRST(extern)         = { "extern" }

FIRST(decl_list)      = { "void", "int", "float", "bool" }
FIRST(decl_list')     = { "void", "int", "float", "bool", ε }

FIRST(var_decl)       = { "int", "float", "bool" }
FIRST(type_spec)      = { "void", "int", "float", "bool" }
FIRST(var_type)       = { "int", "float", "bool" }
FIRST(arr_type)       = { "int", "float" }

FIRST(fun_decl)       = { "void", "int", "float", "bool" }

FIRST(params)         = { "int", "float", "bool", "void", ε }
FIRST(param_list)     = { "int", "float", "bool" }
FIRST(param_list')    = { ",", ε }

FIRST(block)          = { "{" }
FIRST(local_decls)    = { "int", "float", "bool", ε }

FIRST(local_decl)     = { "int", "float", "bool" }

FIRST(array_suffix)   = { "[" }

FIRST(stmt_list)      = { "{", "if", "while", "return", IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT, ";", ε }
FIRST(stmt_list')     = { "{", "if", "while", "return", IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT, ";", ε }
FIRST(stmt)           = { "{", "if", "while", "return", IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT, ";" }

FIRST(or_expr)        = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(or_tail)        = { "||", ε }

FIRST(and_expr)       = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(and_tail)       = { "&&", ε }

FIRST(eq_expr)        = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }
FIRST(eq_tail)        = { "==", "!=", ε }

FIRST(rel_expr)       = { IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT }

======================================================
PART C — FOLLOW SETS (headline & ε-nonterminals)
======================================================

FOLLOW(program)       = { $ }

# Lists / tails around top-level structure
FOLLOW(extern_list)   = { "void", "int", "float", "bool" }
FOLLOW(extern_list')  = { "void", "int", "float", "bool" }
FOLLOW(decl_list)     = { $ }
FOLLOW(decl_list')    = { $ }

# Params / args
FOLLOW(params)        = { ")" }
FOLLOW(param_list)    = { ")" }
FOLLOW(param_list')   = { ")" }

FOLLOW(args)          = { ")" }
FOLLOW(arg_list)      = { ")" }
FOLLOW(arg_tail)      = { ")" }

# Blocks & statements
FOLLOW(block)         = { "{", "if", "while", "return", IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT, ";", "}", $ }
FOLLOW(local_decls)   = { "{", "if", "while", "return", IDENT, "-", "!", "(", INT_LIT, FLOAT_LIT, BOOL_LIT, ";", "}" }
FOLLOW(local_decls')  = FOLLOW(local_decls)
FOLLOW(local_decl)    = FOLLOW(local_decls)
FOLLOW(stmt_list)     = { "}" }
FOLLOW(stmt_list')    = { "}" }
FOLLOW(stmt)          = FOLLOW(stmt_list)

# Arrays in declarations
FOLLOW(array_suffix)  = { ";", ")", "," }

# Expression contexts (used in conditions, returns, expr_stmt, args)
FOLLOW(expr)          = { ")", ";", ",", "}" }
FOLLOW(rval)          = FOLLOW(expr)

# Main expression nonterminals
FOLLOW(or_expr)       = FOLLOW(expr)
FOLLOW(or_tail)       = FOLLOW(or_expr)          # = { ")", ";", ",", "}" }

FOLLOW(and_expr)      = { "||" } ∪ FOLLOW(or_expr)
FOLLOW(and_tail)      = { "||" } ∪ FOLLOW(or_expr)

FOLLOW(eq_expr)       = { "&&" } ∪ FOLLOW(and_expr)
FOLLOW(eq_tail)       = { "&&", "||" } ∪ FOLLOW(or_expr)

FOLLOW(rel_expr)      = { "==", "!=" } ∪ FOLLOW(eq_expr)
FOLLOW(rel_tail)      = { "==", "!=", "&&", "||" } ∪ FOLLOW(or_expr)

FOLLOW(add_expr)      = { "<", "<=", ">", ">=" } ∪ FOLLOW(rel_expr)
FOLLOW(add_tail)      = { "<", "<=", ">", ">=", "==", "!=", "&&", "||" } ∪ FOLLOW(or_expr)

FOLLOW(mul_expr)      = { "+", "-" } ∪ FOLLOW(add_expr)
FOLLOW(mul_tail)      = { "+", "-", "<", "<=", ">", ">=", "==", "!=", "&&", "||" } ∪ FOLLOW(or_expr)

FOLLOW(unary)         = FOLLOW(mul_expr)
FOLLOW(primary)       = FOLLOW(unary)

# Call / array postfix after IDENT:
FOLLOW(primary_tail)  = { "*", "/", "%", "+", "-", "<", "<=", ">", ">=", "==", "!=", "&&", "||", ")", ";", ",", "}" }

FOLLOW(var_decl)      = FOLLOW(decl_list)
FOLLOW(local_decl)    = FOLLOW(local_decls)
FOLLOW(param)         = { ")", "," }


======================================================
PART D — QUICK LL(k) CHECK AND DESIGN EXPLANATION
======================================================

In this section I explain in first person what I changed and why the result remains LL(k) and matches the array task.

1. Arrays only for int and float

I keep scalar types as:

    var_type ::= "int" | "float" | "bool"

Then I introduce:

    arr_type ::= "int" | "float"

and use `arr_type` only for array declarations:

    var_decl   ::= var_type IDENT ";"
                 | arr_type IDENT array_suffix ";"

    local_decl ::= var_type IDENT ";"
                 | arr_type IDENT array_suffix ";"

    param      ::= var_type IDENT
                 | arr_type IDENT array_suffix

This way I still allow `bool x;` as a scalar, but I disallow `bool a[5];` at grammar level because `arr_type` does not include `"bool"`. That matches the requirement that arrays are only for `int` and `float`.

2. Fixed 1D, 2D, 3D arrays with explicit sizes

To enforce that every dimension has a size and that I only support 1D, 2D, and 3D, I define:

    array_suffix ::= "[" INT_LIT "]"
                   | "[" INT_LIT "]" "[" INT_LIT "]"
                   | "[" INT_LIT "]" "[" INT_LIT "]" "[" INT_LIT "]"

So a declaration like:

    int arr[5];
    int arr2D[4][4];
    float arr3D[3][3][4];

matches the productions, and patterns like `int a[];` or `int a[2][2][2][2];` do not appear in the language.

3. Array access at the same precedence level as function calls

The task asks for array element access at the same precedence level as function calls. I already have:

    primary ::= "(" expr ")"
              | IDENT primary_tail
              | INT_LIT
              | FLOAT_LIT
              | BOOL_LIT

and I extend `primary_tail` to hold both function calls and array accesses:

    primary_tail ::= "(" args ")" primary_tail
                   | "[" expr "]" primary_tail
                   | ε

So once the parser has seen an `IDENT`, it decides among:

- `"("` for a function call,
- `"["` for an array access,
- or a token in FOLLOW(primary_tail) to end the postfix chain.

Because `primary_tail` recurs on itself I support sequences such as:

- `arr[i][j]`
- `foo(a[2])`

and function calls and array accesses share the same precedence position.

4. LL(k) decisions and FIRST/FOLLOW consistency

The main LL(2) decision in the original grammar remains:

    expr ::= IDENT "=" expr
           | rval

When I see `IDENT` at the start of `expr`, I use one token of lookahead:

- `"="` means the assignment alternative,
- anything else means I treat the `IDENT` as part of `rval` (through `primary`, `or_expr`, and so on).

For declarations, I now distinguish scalar and array forms after reading the type and identifier. For example:

    var_decl ::= var_type IDENT ";"
               | arr_type IDENT array_suffix ";"

After `"int" IDENT`, I look ahead:

- `";"` means `int x;` (scalar),
- `"["` means `int a[5];` (array with `array_suffix`).

This still uses bounded lookahead, so the grammar remains LL(k) for a small fixed k that matches the coursework allowance.

For `primary_tail`, the decision is LL(1) because:

- FIRST(primary_tail) = { "(", "[", ε }
- FOLLOW(primary_tail) = { "*", "/", "%", "+", "-", "<", "<=", ">", ">=", "==", "!=", "&&", "||", ")", ";", ",", "}" }

The operator tokens and closing tokens only appear in FOLLOW, not in FIRST, so if the next token is:

- `"("` I parse a function call tail,
- `"["` I parse an array access tail,
- anything in FOLLOW(primary_tail) I choose ε.

5. Link to later tasks (parser, AST, zero initialisation)

This grammar lines up with the next task where I extend the recursive descent parser and AST:

- The lexer gains tokens `[` and `]`.
- In parsing, `var_decl`, `local_decl`, and `param` need to call a helper for `array_suffix` when they see `[` after an identifier.
- In expressions, `parsePrimary` or `parsePrimaryTail` needs to recognise `[` after an `IDENT` and build an array access AST node.
- I plan to create AST nodes such as `ArrayDecl` (with element type, name, and dimension sizes) and `ArrayAccess` (with a base expression and index expressions).
- The assumption that arrays are initialised to `0` or `0.0` belongs to semantic analysis or code generation, not the grammar, so I enforce it there by emitting the right LLVM IR initialisation.
